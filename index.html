import urllib.parse import urllib.request import time import os import random import sys import json os.makedirs("chunky_soldiers", exist_ok=True) TRACKER_FILE = "chunky_soldiers/generated_prompts.json" def log(message): print(message) sys.stdout.flush() # 1. Load history generated_history = [] if os.path.exists(TRACKER_FILE): try: with open(TRACKER_FILE, 'r') as f: generated_history = json.load(f) except: generated_history = [] # 2. Random variables to force the AI to be creative themes = [ "doing office paperwork", "in a wild west saloon", "in outer space", "at a fancy restaurant", "playing a sport", "in a horror movie", "at the beach", "doing extreme sports", "in a fantasy castle", "in a fast food drive-thru", "at a fancy wedding", "in a zoo", "fighting a boss in a video game", "in a yoga class" ] actions = [ "squeezing into", "falling off", "eating", "hugging", "running away from", "hiding inside", "arguing with", "riding", "tangled in", "lifting", "stuck under", "staring confused at" ] objects = [ "a tiny shopping cart", "a giant rubber duck", "a very small bicycle", "a massive hamburger", "a fragile antique chair", "a tiny clown car", "a giant birthday cake", "a very angry goose", "a piece of gym equipment", "a highly classified military briefcase", "a children's playground slide" ] NUM_IMAGES = 10 log(f"Starting generation of {NUM_IMAGES} AI-generated images...") for i in range(NUM_IMAGES): # Pick random constraints theme = random.choice(themes) action = random.choice(actions) obj = random.choice(objects) log(f"\n[{i+1}/{NUM_IMAGES}] Constraint: {theme} | {action} | {obj}") log("Asking Text AI for a new prompt...") # 3. The chaotic system prompt SYSTEM_PROMPT = ( f"You are a hilarious meme generator. Write ONE short, highly specific image prompt. " f"The image must feature an extremely obese fat man in a military camo uniform. " f"He must be {theme}, specifically {action} {obj}. " f"It must be a funny stock photo style, isolated on a pure white background. " f"Do not include any intro text, just output the prompt directly." ) try: text_api_url = f"https://text.pollinations.ai/{urllib.parse.quote(SYSTEM_PROMPT)}" req_text = urllib.request.urlopen(text_api_url, timeout=20) ai_prompt = req_text.read().decode('utf-8').strip() ai_prompt = ai_prompt.replace('"', '').replace('\n', ' ') log(f"🧠 Text AI generated prompt: {ai_prompt}") if ai_prompt in generated_history: log("⚠️ Duplicate prompt generated by AI. Skipping.") continue encoded_prompt = urllib.parse.quote(ai_prompt) negative_prompt = urllib.parse.quote("woman, girl, female, vehicle, car, ugly, deformed, text, watermark, cartoon, illustration") seed = random.randint(1, 999999) image_url = f"https://image.pollinations.ai/prompt/{encoded_prompt}?width=1024&height=1024&nologo=true&model=flux&negative_prompt={negative_prompt}&seed={seed}" filename = f"chunky_soldiers/soldier_{int(time.time())}_{random.randint(1000, 9999)}.jpg" log("📸 Generating image...") req_img = urllib.request.urlopen(image_url, timeout=30) with open(filename, 'wb') as f: f.write(req_img.read()) log(f"✅ Saved: {filename}") generated_history.append(ai_prompt) with open(TRACKER_FILE, 'w') as f: json.dump(generated_history, f, indent=2) except Exception as e: log(f"❌ Error: {e}") log("Skipping this one...") if i < NUM_IMAGES - 1: log("Waiting 30 seconds to avoid the rate limit...") time.sleep(30) log("\nDone! Run complete. Your army grows.")
